# Shoprelle — https://shoprelle.com

User-agent: *
Allow: /

# Le back-office et les pages de compte n'ont rien à faire dans un index :
# elles exigent une session, et un moteur qui les explore ne récolte que des
# redirections vers la page de connexion.
Disallow: /admin
Disallow: /dashboard
Disallow: /settings
Disallow: /telegram

# La barre finale n'est pas une coquille : /mes-demandes est le formulaire, que
# n'importe qui peut ouvrir et qu'un client qui revient a le droit de trouver.
# Ce qui est derrière — une demande par adresse — n'appartient qu'à son client.
Disallow: /mes-demandes/

Sitemap: https://shoprelle.com/sitemap.xml
